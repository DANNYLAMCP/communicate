    class SoundEngine {
      constructor() {
        this.ctx = null;
        this.masterGain = null;
        this.bgmGain = null;
        this.sfxGain = null;
        this.bgmInterval = null;
        this.isPlayingBGM = false;
        this.isMuted = false;
        this.lastVolume = 0.5;
      }

      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext  window.webkitAudioContext;
          this.ctx = new AudioContext();

           主音量 Control
          this.masterGain = this.ctx.createGain();
          this.masterGain.gain.setValueAtTime(0.5, this.ctx.currentTime);
          this.masterGain.connect(this.ctx.destination);

           BGM 分道 Gain
          this.bgmGain = this.ctx.createGain();
          this.bgmGain.gain.setValueAtTime(0.15, this.ctx.currentTime);
          this.bgmGain.connect(this.masterGain);

           SFX 分道 Gain
          this.sfxGain = this.ctx.createGain();
          this.sfxGain.gain.setValueAtTime(1.0, this.ctx.currentTime);
          this.sfxGain.connect(this.masterGain);
        }

        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

       1. 關卡背景音樂 (BGM)
      startBGM() {
        this.init();
        if (this.isPlayingBGM) return;
        this.isPlayingBGM = true;

        const melody = [261.63, 329.63, 392.00, 523.25, 392.00, 329.63];  C4, E4, G4, C5, G4, E4
        let noteIndex = 0;

        this.bgmInterval = setInterval(() = {
          if (!this.isPlayingBGM) return;

          const osc = this.ctx.createOscillator();
          const noteGain = this.ctx.createGain();

          osc.type = 'triangle';
          osc.frequency.setValueAtTime(melody[noteIndex % melody.length], this.ctx.currentTime);

          noteGain.gain.setValueAtTime(0.2, this.ctx.currentTime);
          noteGain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.25);

          osc.connect(noteGain);
          noteGain.connect(this.bgmGain);

          osc.start();
          osc.stop(this.ctx.currentTime + 0.25);

          noteIndex++;
        }, 280);
      }

      stopBGM() {
        this.isPlayingBGM = false;
        if (this.bgmInterval) {
          clearInterval(this.bgmInterval);
        }
      }

       2. 爆炸音效 (White Noise + Lowpass Filter)
      playExplosionSound() {
        this.init();

        const duration = 0.5;
        const bufferSize = this.ctx.sampleRate  duration;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);

        for (let i = 0; i  bufferSize; i++) {
          data[i] = Math.random()  2 - 1;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.setValueAtTime(1000, this.ctx.currentTime);
        filter.frequency.exponentialRampToValueAtTime(30, this.ctx.currentTime + duration);

        const expGain = this.ctx.createGain();
        expGain.gain.setValueAtTime(0.8, this.ctx.currentTime);
        expGain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + duration);

        noise.connect(filter);
        filter.connect(expGain);
        expGain.connect(this.sfxGain);

        noise.start();
      }

       3. 機制 4 閃燈音效 (Upward Pitch Bend)
      playBlinkSound() {
        this.init();

        const osc = this.ctx.createOscillator();
        const noteGain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(880, this.ctx.currentTime);  A5
        osc.frequency.exponentialRampToValueAtTime(1760, this.ctx.currentTime + 0.12);  A6

        noteGain.gain.setValueAtTime(0.3, this.ctx.currentTime);
        noteGain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.12);

        osc.connect(noteGain);
        noteGain.connect(this.sfxGain);

        osc.start();
        osc.stop(this.ctx.currentTime + 0.12);
      }

       設定主音量
      setVolume(val) {
        this.init();
        this.masterGain.gain.setValueAtTime(val, this.ctx.currentTime);
      }
    }

     初始化引擎與頁面邏輯
    const sound = new SoundEngine();

    function enterLevel() {
      sound.startBGM();
      document.getElementById('status').innerText = 🔊 進入關卡：BGM 播放中;
    }

    function triggerExplosion() {
      sound.playExplosionSound();
      const card = document.getElementById('gameCard');
      card.classList.add('shake');
      setTimeout(() = card.classList.remove('shake'), 350);
    }

    function triggerMechanism4() {
      sound.playBlinkSound();
      const lamp = document.getElementById('lamp4');
      lamp.classList.add('active');
      setTimeout(() = lamp.classList.remove('active'), 180);
    }

    function exitLevel() {
      sound.stopBGM();
      document.getElementById('status').innerText = ⏸ 已離開關卡：BGM 停止;
    }

    function changeVolume(val) {
      sound.setVolume(val);
      if (val  0 && sound.isMuted) {
        sound.isMuted = false;
        document.getElementById('muteBtn').innerText = 靜音;
      }
    }

    function toggleMute() {
      const slider = document.getElementById('volumeSlider');
      const muteBtn = document.getElementById('muteBtn');

      if (sound.isMuted) {
        sound.isMuted = false;
        slider.value = sound.lastVolume;
        sound.setVolume(sound.lastVolume);
        muteBtn.innerText = 靜音;
      } else {
        sound.lastVolume = slider.value;
        sound.isMuted = true;
        slider.value = 0;
        sound.setVolume(0);
        muteBtn.innerText = 取消靜音;
      }
    }
